# =============================================================================
# huginn_core_tests -- host tests for src/core/ (see src/core/README.md)
# =============================================================================
# Builds and runs on the host, without the game: no CommonLib, no SKSE, and NOT
# the plugin's precompiled header (PCH.h is attached to the Huginn target only,
# PRIVATE). The plugin compiles the same src/core/ files too (its
# GLOB_RECURSE over src/ picks them up), so the game layer calls this code.
#
# Test sources live here, outside src/, so the plugin's glob never compiles a
# test (or doctest's main) into Huginn.dll.
#
#   ctest -C Debug --test-dir build --output-on-failure
# =============================================================================

set(HUGINN_CORE_DIR "${PROJECT_SOURCE_DIR}/src/core")
set(HUGINN_PURITY_SCRIPT "${PROJECT_SOURCE_DIR}/cmake/CheckCorePurity.cmake")

file(GLOB_RECURSE HUGINN_CORE_FILES CONFIGURE_DEPENDS
   "${HUGINN_CORE_DIR}/*.h" "${HUGINN_CORE_DIR}/*.hpp" "${HUGINN_CORE_DIR}/*.hxx"
   "${HUGINN_CORE_DIR}/*.inl" "${HUGINN_CORE_DIR}/*.ipp"
   "${HUGINN_CORE_DIR}/*.cpp" "${HUGINN_CORE_DIR}/*.cxx" "${HUGINN_CORE_DIR}/*.cc")
file(GLOB_RECURSE HUGINN_CORE_HEADERS CONFIGURE_DEPENDS
   "${HUGINN_CORE_DIR}/*.h" "${HUGINN_CORE_DIR}/*.hpp" "${HUGINN_CORE_DIR}/*.hxx")
file(GLOB_RECURSE HUGINN_CORE_SOURCES CONFIGURE_DEPENDS "${HUGINN_CORE_DIR}/*.cpp")
file(GLOB HUGINN_CORE_TEST_SOURCES CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/core/*.cpp")

# --- The purity rule (cmake/CheckCorePurity.cmake) ---------------------------
# Checked twice: here, so a bad tree fails the configure at once, and as a
# build step that reruns whenever a core file (or the script) changes, so an
# edit made after the configure cannot slip through. Both the test target and
# the plugin depend on that step.
execute_process(
   COMMAND "${CMAKE_COMMAND}" "-DCORE_DIR=${HUGINN_CORE_DIR}" "-DSRC_DIR=${PROJECT_SOURCE_DIR}/src"
           -P "${HUGINN_PURITY_SCRIPT}"
   RESULT_VARIABLE purity_result)
if(NOT purity_result EQUAL 0)
   message(FATAL_ERROR "src/core/ purity check failed (details above)")
endif()

set(HUGINN_PURITY_STAMP "${CMAKE_CURRENT_BINARY_DIR}/core_purity.stamp")
add_custom_command(
   OUTPUT "${HUGINN_PURITY_STAMP}"
   COMMAND "${CMAKE_COMMAND}" "-DCORE_DIR=${HUGINN_CORE_DIR}" "-DSRC_DIR=${PROJECT_SOURCE_DIR}/src"
           -P "${HUGINN_PURITY_SCRIPT}"
   COMMAND "${CMAKE_COMMAND}" -E touch "${HUGINN_PURITY_STAMP}"
   DEPENDS "${HUGINN_PURITY_SCRIPT}" ${HUGINN_CORE_FILES}
   COMMENT "Checking that src/core/ stays pure"
   VERBATIM)
add_custom_target(huginn_core_purity DEPENDS "${HUGINN_PURITY_STAMP}")

# --- Every core header compiles on its own ------------------------------------
# One generated TU per header, including it twice (catches a missing include
# guard) and nothing else, so a header that leans on PCH.h or on what another
# header happened to include first fails here.
set(HUGINN_CORE_HEADER_TUS "")
foreach(header IN LISTS HUGINN_CORE_HEADERS)
   file(RELATIVE_PATH rel "${HUGINN_CORE_DIR}" "${header}")
   string(MAKE_C_IDENTIFIER "${rel}" id)
   set(tu "${CMAKE_CURRENT_BINARY_DIR}/core_headers/standalone_${id}.cpp")
   file(CONFIGURE OUTPUT "${tu}"
      CONTENT "// Generated by tests/CMakeLists.txt: core/${rel} must compile alone.\n#include \"core/${rel}\"\n#include \"core/${rel}\"\n")
   list(APPEND HUGINN_CORE_HEADER_TUS "${tu}")
endforeach()

find_package(doctest CONFIG REQUIRED)

add_executable(huginn_core_tests
   ${HUGINN_CORE_TEST_SOURCES} ${HUGINN_CORE_SOURCES} ${HUGINN_CORE_HEADER_TUS})
source_group("Generated\\standalone headers" FILES ${HUGINN_CORE_HEADER_TUS})
target_compile_features(huginn_core_tests PRIVATE cxx_std_23)
# Only src/ is on the include path, and core headers are included as
# "core/X.h". A test that includes a game header fails to compile here (no
# PCH, no CommonLib), which is the point.
target_include_directories(huginn_core_tests PRIVATE "${PROJECT_SOURCE_DIR}/src")
target_link_libraries(huginn_core_tests PRIVATE doctest::doctest)
# Data-driven tests (the race map) read repo files by absolute path.
target_compile_definitions(huginn_core_tests PRIVATE "HUGINN_REPO_ROOT=\"${PROJECT_SOURCE_DIR}\"")
if(MSVC)
   target_compile_options(huginn_core_tests PRIVATE /utf-8 /permissive- /Zc:preprocessor)
endif()

add_dependencies(huginn_core_tests huginn_core_purity)
if(TARGET Huginn)
   add_dependencies(Huginn huginn_core_purity)
endif()

add_test(NAME huginn_core_tests COMMAND huginn_core_tests)
